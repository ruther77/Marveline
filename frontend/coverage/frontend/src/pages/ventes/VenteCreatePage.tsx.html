import { useState } from 'react'
import { useNavigate } from '@tanstack/react-router'
import { ArrowLeft, Plus, Trash2, ShoppingBag } from 'lucide-react'
import { useCreateVente } from '@/api/queries/useVentes'
import { useCustomersList } from '@/api/queries/useCustomers'
import { CataloguePickerModal } from '@/components/catalogue/CataloguePickerModal'
import { formatCents } from '@/lib/utils'
import type { CataloguePickerLine } from '@/components/catalogue/CataloguePickerModal'
 
type Line = {
  key: number
  product_id: number
  product_name: string
  quantity: number
  unit_price_cents: number
}
 
let lineKey = 0
 
export default function VenteCreatePage() {
  const navigate = useNavigate()
  const createVente = useCreateVente()
  const { data: customersData } = useCustomersList({ page_size: 200 })
  const customers = customersData?.items ?? []
 
  const [customerId, setCustomerId] = useState<number | ''>('')
  const [saleDate, setSaleDate] = useState(new Date().toISOString().slice(0, 10))
  const [depositPct, setDepositPct] = useState('')
  const [paymentDueDate, setPaymentDueDate] = useState('')
  const [notes, setNotes] = useState('')
  const [lines, setLines] = useState<Line[]>([])
  const [pickerOpen, setPickerOpen] = useState(false)
  const [error, setError] = useState<string | null>(null)
 
  const handlePickLine = (l: CataloguePickerLine) => {
    setLines((prev) => {
      const existing = prev.find((li) => li.product_id === l.product_id)
      if (existing) {
        return prev.map((li) =>
          li.product_id === l.product_id
            ? { ...li, quantity: li.quantity + l.quantity, unit_price_cents: l.unit_price_cents }
            : li
        )
      }
      return [...prev, { key: ++lineKey, ...l, product_name: l.product_name ?? `Produit #${l.product_id}` }]
    })
    setPickerOpen(false)
  }
 
  const removeLine = (key: number) => setLines((prev) => prev.filter((l) => l.key !== key))
 
  const updateQty = (key: number, qty: number) => {
    if (qty <= 0) { removeLine(key); return }
    setLines((prev) => prev.map((l) => l.key === key ? { ...l, quantity: qty } : l))
  }
 
  const totalCents = lines.reduce((sum, l) => sum + l.quantity * l.unit_price_cents, 0)
 
  const handleSubmit = () => {
    if (!customerId) { setError('Sélectionner un client'); return }
    if (lines.length === 0) { setError('Ajouter au moins un article'); return }
    setError(null)
 
    createVente.mutate(
      {
        customer_id: customerId as number,
        sale_date: saleDate,
        lines: lines.map((l) => ({
          product_id: l.product_id,
          quantity: l.quantity,
          unit_price_cents: l.unit_price_cents,
        })),
        deposit_pct: depositPct ? parseFloat(depositPct) : undefined,
        payment_due_date: paymentDueDate || undefined,
        notes: notes.trim() || undefined,
      },
      {
        onSuccess: (data) => navigate({ to: '/ventes/$id', params: { id: String(data.id) } }),
        onError: (err) => {
          setError(
            (err as { response?: { data?: { detail?: string } } })?.response?.data?.detail ||
              'Erreur lors de la création.'
          )
        },
      }
    )
  }
 
  return (
    <div className="p-4 md:p-6 max-w-2xl mx-auto space-y-5">
      {/* Header */}
      <div className="flex items-center gap-3">
        <button
          onClick={() => navigate({ to: '/ventes' })}
          className="p-2 hover:bg-dark-700 rounded text-dark-400"
        >
          <ArrowLeft className="w-5 h-5" />
        </button>
        <div className="flex items-center gap-2">
          <ShoppingBag className="w-5 h-5 text-gold-400" />
          <h1 className="text-xl font-semibold text-white">Nouvelle vente</h1>
        </div>
      </div>
 
      {/* Client + Date */}
      <div className="card space-y-4">
        <h2 className="text-sm font-medium text-dark-300">Informations</h2>
        <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
          <div>
            <label className="block text-sm text-dark-400 mb-1">Client *</label>
            <select
              value={customerId}
              onChange={(e) => setCustomerId(e.target.value ? Number(e.target.value) : '')}
              className="w-full bg-dark-700 border border-dark-600 rounded-lg px-3 py-2 text-white text-sm focus:outline-none focus:border-gold-500"
            >
              <option value="">Sélectionner un client…</option>
              {customers.map((c) => (
                <option key={c.id} value={c.id}>{c.first_name} {c.last_name}</option>
              ))}
            </select>
          </div>
          <div>
            <label className="block text-sm text-dark-400 mb-1">Date de vente *</label>
            <input
              type="date"
              value={saleDate}
              onChange={(e) => setSaleDate(e.target.value)}
              className="w-full bg-dark-700 border border-dark-600 rounded-lg px-3 py-2 text-white text-sm focus:outline-none focus:border-gold-500"
            />
          </div>
          <div>
            <label className="block text-sm text-dark-400 mb-1">Acompte (%)</label>
            <input
              type="number"
              step="5"
              min={0}
              max={100}
              placeholder="ex : 30"
              value={depositPct}
              onChange={(e) => setDepositPct(e.target.value)}
              className="w-full bg-dark-700 border border-dark-600 rounded-lg px-3 py-2 text-white text-sm focus:outline-none focus:border-gold-500"
            />
          </div>
          <div>
            <label className="block text-sm text-dark-400 mb-1">Échéance de paiement</label>
            <input
              type="date"
              value={paymentDueDate}
              onChange={(e) => setPaymentDueDate(e.target.value)}
              className="w-full bg-dark-700 border border-dark-600 rounded-lg px-3 py-2 text-white text-sm focus:outline-none focus:border-gold-500"
            />
          </div>
        </div>
        <div>
          <label className="block text-sm text-dark-400 mb-1">Notes (optionnel)</label>
          <textarea
            rows={2}
            placeholder="Informations complémentaires…"
            value={notes}
            onChange={(e) => setNotes(e.target.value)}
            className="w-full bg-dark-700 border border-dark-600 rounded-lg px-3 py-2 text-white text-sm focus:outline-none focus:border-gold-500 resize-none"
          />
        </div>
      </div>
 
      {/* Articles */}
      <div className="card space-y-3">
        <div className="flex items-center justify-between">
          <h2 className="text-sm font-medium text-dark-300">Articles</h2>
          <button
            onClick={() => setPickerOpen(true)}
            className="flex items-center gap-1.5 text-xs text-gold-400 hover:text-gold-300"
          >
            <Plus className="w-3.5 h-3.5" />
            Ajouter un article
          </button>
        </div>
 
        {lines.length === 0 ? (
          <button
            onClick={() => setPickerOpen(true)}
            className="w-full border-2 border-dashed border-dark-600 hover:border-gold-600 rounded-xl py-6 text-dark-400 hover:text-gold-400 text-sm transition-colors"
          >
            + Sélectionner des articles depuis le catalogue
          </button>
        ) : (
          <div className="space-y-2">
            {lines.map((l) => (
              <div key={l.key} className="flex items-center gap-3 text-sm">
                <div className="flex-1">
                  <p className="text-white">{l.product_name}</p>
                  <p className="text-dark-400 text-xs">{formatCents(l.unit_price_cents)} / unité</p>
                </div>
                <input
                  type="number"
                  min={1}
                  value={l.quantity}
                  onChange={(e) => updateQty(l.key, parseInt(e.target.value) || 0)}
                  className="w-16 bg-dark-700 border border-dark-600 rounded px-2 py-1 text-white text-sm text-center focus:outline-none focus:border-gold-500"
                />
                <p className="text-white font-medium w-20 text-right">{formatCents(l.quantity * l.unit_price_cents)}</p>
                <button onClick={() => removeLine(l.key)} className="text-dark-500 hover:text-red-400 p-1">
                  <Trash2 className="w-3.5 h-3.5" />
                </button>
              </div>
            ))}
            <div className="border-t border-dark-700 pt-3 flex justify-between font-semibold text-sm">
              <span className="text-dark-300">Total estimé</span>
              <span className="text-gold-400">{formatCents(totalCents)}</span>
            </div>
          </div>
        )}
      </div>
 
      {error && (
        <div className="bg-red-900/20 border border-red-700 rounded-lg px-4 py-3 text-red-400 text-sm">
          {error}
        </div>
      )}
 
      {/* Submit */}
      <button
        onClick={handleSubmit}
        disabled={!customerId || lines.length === 0 || createVente.isPending}
        className="w-full flex items-center justify-center gap-2 bg-gold-500 hover:bg-gold-600 text-dark-900 font-semibold py-3 rounded-xl disabled:opacity-50"
      >
        <ShoppingBag className="w-5 h-5" />
        {createVente.isPending ? 'Création…' : 'Créer la vente'}
      </button>
 
      <CataloguePickerModal
        isOpen={pickerOpen}
        onClose={() => setPickerOpen(false)}
        onSelect={handlePickLine}
      />
    </div>
  )
}
