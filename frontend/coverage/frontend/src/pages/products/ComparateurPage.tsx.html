import { useState } from 'react'
import { GitCompare, X, Search, Check } from 'lucide-react'
import { useProductsList } from '@/api/queries'
import type { Product } from '@/types/product'
 
const CONDITION_LABELS: Record<string, string> = {
  new: 'Neuf',
  good: 'Bon état',
  fair: 'Correct',
  poor: 'Usé',
}
 
const ROWS: { label: string; key: keyof Product | 'deposit' }[] = [
  { label: 'Référence (SKU)', key: 'sku' },
  { label: 'Catégorie', key: 'category' },
  { label: 'Prix / jour', key: 'price_per_day_euros' },
  { label: 'Stock total', key: 'stock_quantity' },
  { label: 'Disponible', key: 'available_quantity' },
  { label: 'État', key: 'condition' },
  { label: 'Caution', key: 'deposit' },
]
 
function cellValue(p: Product, key: typeof ROWS[0]['key']): string {
  if (key === 'deposit') {
    const euros = (p as Product & { deposit_amount_euros?: number }).deposit_amount_euros
    return euros != null ? `${euros.toFixed(2)} €` : '—'
  }
  if (key === 'price_per_day_euros') {
    const v = p[key as keyof Product]
    return v != null ? `${Number(v).toFixed(2)} €` : '—'
  }
  if (key === 'condition') return CONDITION_LABELS[p.condition ?? ''] ?? p.condition ?? '—'
  if (key === 'category') return (p.category as { name?: string } | null)?.name ?? '—'
  const v = p[key as keyof Product]
  return v != null ? String(v) : '—'
}
 
export default function ComparateurPage() {
  const [search, setSearch] = useState('')
  const [selected, setSelected] = useState<Product[]>([])
 
  const { data, isLoading } = useProductsList({ page_size: 200, search: search || undefined })
  const products: Product[] = data?.items ?? []
 
  const toggle = (p: Product) => {
    setSelected((prev) => {
      if (prev.find((x) => x.id === p.id)) return prev.filter((x) => x.id !== p.id)
      if (prev.length >= 4) return prev
      return [...prev, p]
    })
  }
 
  const isSelected = (p: Product) => selected.some((x) => x.id === p.id)
 
  return (
    <div className="p-4 md:p-6 space-y-5">
      <div className="flex items-center gap-3">
        <GitCompare className="w-5 h-5 text-gold-400" />
        <h1 className="text-xl font-semibold text-white">Comparateur produits</h1>
        <span className="text-xs text-dark-400 ml-2">Sélectionnez jusqu'à 4 produits</span>
      </div>
 
      {/* Produits sélectionnés — chips */}
      {selected.length > 0 && (
        <div className="flex flex-wrap gap-2">
          {selected.map((p) => (
            <span
              key={p.id}
              className="flex items-center gap-1.5 px-3 py-1 bg-gold-500/20 text-gold-300 border border-gold-500/40 rounded-full text-sm"
            >
              {p.name}
              <button onClick={() => toggle(p)} className="hover:text-gold-100">
                <X className="w-3.5 h-3.5" />
              </button>
            </span>
          ))}
        </div>
      )}
 
      <div className="grid md:grid-cols-[320px_1fr] gap-5">
        {/* Panneau de sélection */}
        <div className="card p-0 overflow-hidden flex flex-col max-h-[600px]">
          <div className="p-3 border-b border-dark-700">
            <div className="relative">
              <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-dark-400" />
              <input
                type="text"
                value={search}
                onChange={(e) => setSearch(e.target.value)}
                placeholder="Rechercher un produit…"
                className="w-full pl-9 pr-3 py-2 bg-dark-700 text-white text-sm rounded-lg border border-dark-600 focus:outline-none focus:border-gold-500/60 placeholder:text-dark-500"
              />
            </div>
          </div>
          <div className="overflow-y-auto flex-1 divide-y divide-dark-700">
            {isLoading ? (
              <div className="p-8 text-center text-dark-400 text-sm">Chargement…</div>
            ) : products.length === 0 ? (
              <div className="p-8 text-center text-dark-400 text-sm">Aucun résultat</div>
            ) : (
              products.map((p) => {
                const sel = isSelected(p)
                const disabled = !sel && selected.length >= 4
                return (
                  <button
                    key={p.id}
                    onClick={() => !disabled && toggle(p)}
                    className={`w-full flex items-center gap-3 px-4 py-3 text-left transition-colors ${
                      sel
                        ? 'bg-gold-500/10'
                        : disabled
                        ? 'opacity-40 cursor-not-allowed'
                        : 'hover:bg-dark-700'
                    }`}
                  >
                    <div
                      className={`w-5 h-5 rounded border flex items-center justify-center shrink-0 ${
                        sel ? 'bg-gold-500 border-gold-500' : 'border-dark-500'
                      }`}
                    >
                      {sel && <Check className="w-3 h-3 text-dark-900" />}
                    </div>
                    <div className="min-w-0">
                      <p className="text-white text-sm font-medium truncate">{p.name}</p>
                      <p className="text-dark-400 text-xs truncate">{p.sku}</p>
                    </div>
                  </button>
                )
              })
            )}
          </div>
        </div>
 
        {/* Tableau de comparaison */}
        <div>
          {selected.length < 2 ? (
            <div className="card flex items-center justify-center h-48">
              <p className="text-dark-400 text-sm">Sélectionnez au moins 2 produits pour comparer</p>
            </div>
          ) : (
            <div className="card p-0 overflow-x-auto">
              <table className="w-full text-sm">
                <thead>
                  <tr className="border-b border-dark-700">
                    <th className="text-left px-4 py-3 text-dark-400 font-medium w-36">Critère</th>
                    {selected.map((p) => (
                      <th key={p.id} className="px-4 py-3 text-center min-w-[160px]">
                        <div className="flex flex-col items-center gap-1">
                          {p.image_url ? (
                            <img src={p.image_url} alt={p.name} className="w-10 h-10 object-cover rounded" />
                          ) : (
                            <div className="w-10 h-10 bg-dark-700 rounded flex items-center justify-center">
                              <span className="text-dark-500 text-lg">📦</span>
                            </div>
                          )}
                          <span className="text-white font-medium text-xs text-center leading-tight">{p.name}</span>
                        </div>
                      </th>
                    ))}
                  </tr>
                </thead>
                <tbody>
                  {ROWS.map((row, i) => (
                    <tr key={row.key} className={i % 2 === 0 ? 'bg-dark-800/40' : ''}>
                      <td className="px-4 py-2.5 text-dark-400 font-medium">{row.label}</td>
                      {selected.map((p) => (
                        <td key={p.id} className="px-4 py-2.5 text-center text-white">
                          {cellValue(p, row.key)}
                        </td>
                      ))}
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>
          )}
        </div>
      </div>
    </div>
  )
}
