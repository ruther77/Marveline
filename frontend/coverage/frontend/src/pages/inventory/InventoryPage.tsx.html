import { useState } from 'react'
import { useQueries, useQueryClient } from '@tanstack/react-query'
import { Link } from '@tanstack/react-router'
import { useProductsList } from '@/api/queries'
import { inventoryApi } from '@/api/inventory'
import { Search, Package, AlertCircle, CheckCircle, History } from 'lucide-react'
import { cn } from '@/lib/utils'
import { getStockStatus } from '@/types/product'
import type { ProductStockDetail } from '@/types'
import { STOCK_ITEM_STATUS_LABELS as STATUS_LABELS, STOCK_ITEM_STATUS_COLORS as STATUS_COLORS } from '@/lib/constants'
 
type StatusKey = keyof typeof STATUS_LABELS
 
const ALL_STATUSES: StatusKey[] = ['available', 'reserved', 'on_location', 'damaged', 'in_repair']
 
// Maximum products for which we load stock detail (avoid N+1 on full catalog)
const STOCK_DETAIL_LIMIT = 50
 
export default function InventoryPage() {
  const [search, setSearch] = useState('')
  const [editingItem, setEditingItem] = useState<{ productId: number; itemId: number } | null>(null)
  const qc = useQueryClient()
  const { data, isLoading } = useProductsList({ page: 1, page_size: 1000, active_only: true })
 
  const handleStatusChange = async (productId: number, itemId: number, status: string) => {
    try {
      await inventoryApi.patchStockItemStatus(productId, itemId, status)
      qc.invalidateQueries({ queryKey: ['product-stock', productId] })
    } catch {
      // ignore, user sees no change
    } finally {
      setEditingItem(null)
    }
  }
 
  const products = data?.items || []
 
  const filteredProducts = search
    ? products.filter(
        (p) =>
          p.name.toLowerCase().includes(search.toLowerCase()) ||
          p.sku.toLowerCase().includes(search.toLowerCase())
      )
    : products
 
  // Only fetch stock detail when the filtered set is small enough
  const productsForDetail = filteredProducts.slice(0, STOCK_DETAIL_LIMIT)
  const shouldFetchDetail = filteredProducts.length <= STOCK_DETAIL_LIMIT
 
  const stockQueries = useQueries({
    queries: productsForDetail.map((p) => ({
      queryKey: ['product-stock', p.id],
      queryFn: () => inventoryApi.getProductStock(p.id),
      staleTime: 60_000,
      enabled: shouldFetchDetail,
    })),
  })
 
  const stockDetailMap: Record<number, ProductStockDetail> = {}
  productsForDetail.forEach((p, i) => {
    const result = stockQueries[i]
    if (result?.data) {
      stockDetailMap[p.id] = result.data
    }
  })
 
  // Summary stats using product list data
  const totalAvailable = products.reduce((sum, p) => sum + p.available_quantity, 0)
  const totalStock = products.reduce((sum, p) => sum + p.stock_quantity, 0)
  const lowStock = products.filter((p) => getStockStatus(p) === 'low_stock').length
  const outStock = products.filter((p) => getStockStatus(p) === 'out_of_stock').length
 
  const getStockIcon = (status: string) => {
    switch (status) {
      case 'in_stock':
        return <CheckCircle className="w-5 h-5 text-green-500" />
      case 'low_stock':
        return <AlertCircle className="w-5 h-5 text-yellow-500" />
      case 'out_of_stock':
        return <AlertCircle className="w-5 h-5 text-red-500" />
      default:
        return <Package className="w-5 h-5 text-dark-400" />
    }
  }
 
  return (
    <div className="space-y-6">
      <div>
        <h1 className="text-2xl font-bold">Inventaire & Stock</h1>
        <p className="text-dark-400 mt-1">Vue d'ensemble du stock disponible</p>
      </div>
 
      {/* Stats */}
      <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
        <div className="card">
          <div className="flex items-center justify-between">
            <div>
              <p className="text-sm text-dark-400">Unités disponibles</p>
              <p className="text-2xl font-bold text-green-500 mt-1">{totalAvailable}</p>
              <p className="text-xs text-dark-500 mt-0.5">sur {totalStock} en stock</p>
            </div>
            <CheckCircle className="w-8 h-8 text-green-500" />
          </div>
        </div>
 
        <div className="card">
          <div className="flex items-center justify-between">
            <div>
              <p className="text-sm text-dark-400">Stock faible</p>
              <p className="text-2xl font-bold text-yellow-500 mt-1">{lowStock}</p>
              <p className="text-xs text-dark-500 mt-0.5">référence{lowStock !== 1 ? 's' : ''}</p>
            </div>
            <AlertCircle className="w-8 h-8 text-yellow-500" />
          </div>
        </div>
 
        <div className="card">
          <div className="flex items-center justify-between">
            <div>
              <p className="text-sm text-dark-400">Rupture</p>
              <p className="text-2xl font-bold text-red-500 mt-1">{outStock}</p>
              <p className="text-xs text-dark-500 mt-0.5">référence{outStock !== 1 ? 's' : ''}</p>
            </div>
            <AlertCircle className="w-8 h-8 text-red-500" />
          </div>
        </div>
      </div>
 
      {/* Search */}
      <div className="card">
        <div className="relative">
          <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-5 h-5 text-dark-400" />
          <input
            type="text"
            placeholder="Rechercher un produit..."
            value={search}
            onChange={(e) => setSearch(e.target.value)}
            className="input pl-10"
          />
        </div>
        {!search && products.length > STOCK_DETAIL_LIMIT && (
          <p className="mt-2 text-xs text-dark-500">
            Recherchez pour afficher le détail des 5 statuts par produit
          </p>
        )}
      </div>
 
      {/* Products Grid */}
      <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
        {isLoading ? (
          <div className="col-span-full text-center py-8 text-dark-400">
            Chargement...
          </div>
        ) : filteredProducts.length === 0 ? (
          <div className="col-span-full text-center py-8 text-dark-400">
            Aucun produit trouvé
          </div>
        ) : (
          filteredProducts.map((product) => {
            const stockStatus = getStockStatus(product)
            const detail = stockDetailMap[product.id]
            return (
              <div key={product.id} className="card">
                <div className="flex items-start gap-4">
                  <div className="p-3 bg-dark-800 rounded-lg">
                    {getStockIcon(stockStatus)}
                  </div>
 
                  <div className="flex-1 min-w-0">
                    <h3 className="font-medium truncate">{product.name}</h3>
                    <p className="text-sm text-dark-400 font-mono">{product.sku}</p>
 
                    <div className="mt-3 flex flex-wrap gap-1.5">
                      {detail ? (
                        <>
                          {(['available', 'reserved', 'on_location', 'damaged', 'in_repair'] as StatusKey[]).map(
                            (status) => {
                              const count = detail[`qty_${status}` as keyof ProductStockDetail] as number
                              if (count === 0) return null
                              return (
                                <span
                                  key={status}
                                  className={cn(
                                    'inline-flex items-center px-2 py-0.5 rounded-full text-xs font-medium',
                                    STATUS_COLORS[status]
                                  )}
                                >
                                  {count} {STATUS_LABELS[status]}
                                </span>
                              )
                            }
                          )}
                          {detail.qty_available === 0 &&
                            detail.qty_reserved === 0 &&
                            detail.qty_on_location === 0 &&
                            detail.qty_damaged === 0 &&
                            detail.qty_in_repair === 0 && (
                              <span className="text-xs text-dark-400">Aucune unité</span>
                            )}
                        </>
                      ) : (
                        <span className="text-xs text-green-400">
                          {product.available_quantity} dispo / {product.stock_quantity} total
                        </span>
                      )}
                    </div>
 
                    {/* Individual stock items */}
                    {detail && detail.items.length > 0 && (
                      <div className="mt-3 border-t border-dark-700 pt-2 space-y-1">
                        {detail.items.map((item) => (
                          <div
                            key={item.id}
                            className="flex items-center justify-between py-0.5"
                          >
                            <div className="flex items-center gap-1.5 min-w-0">
                              <span className="text-xs text-dark-500">#{item.id}</span>
                              {item.serial_number && (
                                <span className="text-xs font-mono text-dark-400 truncate">
                                  {item.serial_number}
                                </span>
                              )}
                              {editingItem?.productId === product.id && editingItem?.itemId === item.id ? (
                                <select
                                  autoFocus
                                  className="text-xs bg-dark-800 border border-dark-600 rounded px-1 py-0.5"
                                  defaultValue={item.status}
                                  onBlur={() => setEditingItem(null)}
                                  onChange={(e) => handleStatusChange(product.id, item.id, e.target.value)}
                                >
                                  {ALL_STATUSES.map((s) => (
                                    <option key={s} value={s}>{STATUS_LABELS[s]}</option>
                                  ))}
                                </select>
                              ) : (
                                <button
                                  onClick={() => setEditingItem({ productId: product.id, itemId: item.id })}
                                  title="Modifier le statut"
                                  className={cn(
                                    'inline-flex items-center px-1.5 py-0 rounded text-xs cursor-pointer hover:opacity-80',
                                    STATUS_COLORS[item.status as StatusKey] || 'bg-dark-700 text-dark-400'
                                  )}
                                >
                                  {STATUS_LABELS[item.status as StatusKey] || item.status}
                                </button>
                              )}
                            </div>
                            <Link
                              to="/inventory/stock/$id"
                              params={{ id: `${product.id}-${item.id}` }}
                              className="flex-shrink-0 text-dark-500 hover:text-primary-400 flex items-center gap-0.5 text-xs"
                            >
                              <History className="w-3 h-3" />
                              <span>Détail</span>
                            </Link>
                          </div>
                        ))}
                      </div>
                    )}
 
                    <div className="mt-2 flex items-center justify-between text-sm">
                      <span className="text-dark-400">Prix/jour</span>
                      <span className="font-medium">{product.price_per_day_euros.toFixed(2)} €</span>
                    </div>
                  </div>
                </div>
              </div>
            )
          })
        )}
      </div>
 
    </div>
  )
}
